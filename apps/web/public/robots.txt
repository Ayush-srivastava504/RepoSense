# Locale-prefixed job DETAIL pages (/es/jobs/<slug> ...) render the same English
# job as the unprefixed URL and canonicalise to it, and they are not in any
# sitemap. Every one is a duplicate Googlebot would spend crawl budget on, which

# Prevent crawl budget exhaustion on duplicate query permutations.
# NOTE: ?page= is deliberately NOT blocked. Paginated list pages (12 jobs each) are the
# main crawlable internal-link path to job detail pages, and each ?page=N is
# self-canonical (lib/seo/pagination.ts). /_next/ is also deliberately not blocked:
# Googlebot needs the JS/CSS under /_next/static/ to render pages.
Disallow: /*?*search=

Disallow: /*?*sort=
